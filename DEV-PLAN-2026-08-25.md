### Feature 2: Excel Upload on Tag Locator

Upload an Excel file containing tag EPCs, parse them, and add all as targets.

#### 2a. `tag-locator.page.ts` — add upload method

Add these fields and method:

```typescript
import * as XLSX from 'xlsx';  // add at top

// Add as class fields:
excelFile: File | null = null;

// Add method:
onFileSelected(event: any) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.readAsBinaryString(file);
  reader.onload = () => {
    const wb = XLSX.read(reader.result, { type: 'binary' });
    const sheet = wb.Sheets[wb.SheetNames[0]];
    const rows: any[] = XLSX.utils.sheet_to_json(sheet);
    
    // Try common column names: EPC, epc, TagID, tagid, RFIDCode, rfidcode
    for (const row of rows) {
      const epc = row.EPC || row.epc || row.TagID || row.tagid 
                || row.RFIDCode || row.rfidcode || row.A;
      if (epc) this.addTarget(String(epc).trim(), true);
    }
    this.excelFile = null;
  };
}
```

#### 2b. `tag-locator.page.html` — add upload UI inside `.card`
